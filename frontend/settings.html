// frontend/settings.js import APIClient from './apiClient.js'; const apiBaseUrl = window.location.origin; const api = new APIClient(`${apiBaseUrl}/api/v1`); const authApi = new APIClient(apiBaseUrl); class SettingsPage { constructor() { // DOM Elements this.aiModelSelect = document.getElementById('ai-model'); this.aiTemperatureInput = document.getElementById('ai-temperature'); this.aiTemperatureValueSpan = document.getElementById('ai-temperature-value'); this.promptTextarea = document.getElementById('ai-prompt'); this.greetingTextarea = document.getElementById('ai-greeting'); this.voiceSelect = document.getElementById('ai-voice'); this.myPhoneNumberInput = document.getElementById('my-phone-number'); this.callTimeoutInput = document.getElementById('call-timeout'); this.aiMaxTokensInput = document.getElementById('ai-max-tokens'); this.saveButton = document.getElementById('save-settings-btn'); this.resetButton = document.getElementById('reset-settings-btn'); this.statusMessage = document.getElementById('settings-status-message'); this.checkStatusBtn = document.getElementById('check-status-btn'); this.dbStatusEl = document.getElementById('status-database'); this.twilioStatusEl = document.getElementById('status-twilio'); this.geminiStatusEl = document.getElementById('status-gemini'); this.logoutBtn = document.getElementById('logout-btn'); this.init(); } init() { this.loadSettings(); this.setupEventListeners(); } setupEventListeners() { this.saveButton.addEventListener('click', () => this.saveSettings()); this.resetButton.addEventListener('click', () => this.resetSettings()); this.checkStatusBtn.addEventListener('click', () => this.checkConnectivity()); this.aiTemperatureInput.addEventListener('input', (e) => { this.aiTemperatureValueSpan.textContent = e.target.value; }); this.logoutBtn.addEventListener('click', async (e) => { e.preventDefault(); try { await authApi.post('/auth/logout'); window.location.href = '/login.html'; } catch (error) { console.error('Logout failed:', error); } }); // Enter key to save document.addEventListener('keydown', (e) => { if ((e.ctrlKey || e.metaKey) && e.key === 's') { e.preventDefault(); this.saveSettings(); } }); } async loadSettings() { try { const settings = await api.get('/settings'); this.promptTextarea.value = settings.ai_prompt || ''; this.greetingTextarea.value = settings.ai_greeting || "Hello, you've reached the AI assistant. Please state your name and the reason for your call after the beep."; this.voiceSelect.value = settings.ai_voice || 'Polly.Amy'; this.aiModelSelect.value = settings.ai_model || 'gemini-pro'; this.aiTemperatureInput.value = settings.ai_temperature ?? 0.7; this.aiTemperatureValueSpan.textContent = this.aiTemperatureInput.value; this.aiMaxTokensInput.value = settings.ai_max_tokens ?? 200; this.myPhoneNumberInput.value = settings.my_phone_number || ''; this.callTimeoutInput.value = settings.call_timeout || '10'; } catch (error) { console.error('Failed to load settings:', error); this.showStatus('Error loading settings. Please refresh.', 'error'); } } async saveSettings() { if (!confirm('Are you sure you want to save these settings?')) return; const settingsToSave = { ai_prompt: this.promptTextarea.value.trim(), ai_greeting: this.greetingTextarea.value.trim(), ai_voice: this.voiceSelect.value, ai_model: this.aiModelSelect.value, ai_temperature: parseFloat(this.aiTemperatureInput.value), ai_max_tokens: parseInt(this.aiMaxTokensInput.value, 10), my_phone_number: this.myPhoneNumberInput.value.trim(), call_timeout: parseInt(this.callTimeoutInput.value, 10) }; // Validate phone number if (!/^\+[1-9]\d{1,14}$/.test(settingsToSave.my_phone_number)) { this.showStatus('Please enter a valid phone number in E.164 format (e.g., +14155552671).', 'error'); return; } // Validate timeout if (isNaN(settingsToSave.call_timeout) || settingsToSave.call_timeout < 5 || settingsToSave.call_timeout > 60) { this.showStatus('Call Timeout must be between 5 and 60 seconds.', 'error'); return; } // Validate temperature if (isNaN(settingsToSave.ai_temperature) || settingsToSave.ai_temperature < 0 || settingsToSave.ai_temperature > 1) { this.showStatus('AI Temperature must be between 0.0 and 1.0.', 'error'); return; } // Validate max tokens if (isNaN(settingsToSave.ai_max_tokens) || settingsToSave.ai_max_tokens < 1 || settingsToSave.ai_max_tokens > 2048) { this.showStatus('AI Max Response Tokens must be between 1 and 2048.', 'error'); return; } this.saveButton.disabled = true; this.saveButton.innerHTML = ' Saving...'; this.clearStatus(); try { await api.put('/settings', settingsToSave); this.showStatus('Settings saved successfully!', 'success'); } catch (error) { console.error('Failed to save settings:', error); this.showStatus('Failed to save settings: ' + (error.message || 'Unknown error'), 'error'); } finally { this.saveButton.disabled = false; this.saveButton.innerHTML = ' Save All Settings'; } } async resetSettings() { if (!confirm('Are you sure you want to reset all settings to their default values? This cannot be undone.')) return; this.resetButton.disabled = true; this.saveButton.disabled = true; this.resetButton.innerHTML = ' Resetting...'; this.clearStatus(); try { await api.delete('/settings/all'); this.showStatus('Settings reset to defaults successfully!', 'success'); await this.loadSettings(); } catch (error) { console.error('Failed to reset settings:', error); this.showStatus('Failed to reset settings.', 'error'); } finally { this.resetButton.disabled = false; this.saveButton.disabled = false; this.resetButton.innerHTML = ' Reset Defaults'; } } async checkConnectivity() { this.checkStatusBtn.disabled = true; this.checkStatusBtn.innerHTML = ' Checking...'; const statusElements = { database: this.dbStatusEl, twilio: this.twilioStatusEl, gemini: this.geminiStatusEl, }; Object.values(statusElements).forEach(el => { el.textContent = 'Checking...'; el.className = 'status-badge checking'; }); try { const statuses = await api.get('/health/status'); for (const [service, data] of Object.entries(statuses)) { if (statusElements[service]) { const statusClass = data.status === 'ok' ? 'status-ok' : 'status-error'; statusElements[service].textContent = data.message; statusElements[service].className = `status-badge ${statusClass}`; } } } catch (error) { console.error('Failed to fetch connectivity status:', error); Object.values(statusElements).forEach(el => { el.textContent = 'Failed to get status'; el.className = 'status-badge status-error'; }); } finally { this.checkStatusBtn.disabled = false; this.checkStatusBtn.innerHTML = ' Check Status'; } } showStatus(message, type) { this.statusMessage.textContent = message; this.statusMessage.className = `status-message ${type}`; setTimeout(() => this.clearStatus(), 5000); } clearStatus() { this.statusMessage.textContent = ''; this.statusMessage.className = 'status-message'; } } document.addEventListener('DOMContentLoaded', () => new SettingsPage());
